#include <stdio.h>
int main() {
    int n;
    scanf("%d", &n);
    if (n == 0) return 0;
    int cl = 0;
    if (n == 1) cl = 1;
    
    while (pow(2.0, cl) < n) cl++;
    
    for (int i = 0; i < n; i++) {  
        for (int j = cl - 1; j >= 0; j--) {  
            if (i & (1 << j)) printf("1"); 
            else printf("0");  
        }  
        printf(" ");  
    }
}
