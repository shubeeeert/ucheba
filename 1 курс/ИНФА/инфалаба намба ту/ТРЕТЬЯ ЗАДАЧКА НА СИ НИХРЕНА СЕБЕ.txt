#include <stdio.h>
#include <math.h>
int main() {
    int x, m;
    scanf("%d %d", &x, &m);
    int s2 = m + 1;

    for (int i = 0; i <= m; i++) {
        if (pow(2.0, i) > x) {
            s2 = i - 1;
            break;
        }
    }
    if (s2 <= m) {
        if (m != s2 + 1) {
            for (int i = 0; i < m - s2 - 1; i++) {
                printf("0");
            }
        }
        int c = s2;
        for (int i = s2; i >= 0; i--) {
            if (pow(2.0, i) <= x && x < pow(2.0, (i + 1))) {
                x = x - pow(2.0, i);
                if (c == i) printf("1");
            } else printf("0");
            c--;
        }
    } else printf("ошибка");
    return 0;
}
