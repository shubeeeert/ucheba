#ЗАДАНИЕ 1
#include <stdio.h>
int main() {
	int x, n;
	scanf("%d %d", &x, &n);
	if (-pow(2.0, n-1) > x || pow(2.0, n - 1) <= x) {
	    printf("ошибка");
	    return 0;
	}
	if (x >= 0) {
	    printf("0");
	    for (int i = n-2; i >= 0; i--) {
	        if ((x & (1<<i)) != 0) printf("1");
	        else printf("0");
	    }
	} 
	else {
        printf("1");
        for (int i = n-2; i >= 0; i--) {
	        if ((abs(x+1) & (1<<i)) != 0) printf("0");
	        else printf("1");
	    }
    }
    return 0;
}
