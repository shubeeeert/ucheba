#include <stdio.h>
int main() {
	int x1, x2, n, k = 0;
	scanf("%d %d %d", &x1, &x2, &n);
	int a1[n], a2[n];
	if (-pow(2.0, n-1) > x1 || pow(2.0, n - 1) <= x1) {
	    printf("ошибка");
	    return 0;
	}
	if (-pow(2.0, n-1) > x2 || pow(2.0, n - 1) <= x2) {
	    printf("ошибка");
	    return 0;
	}
	if (x1 >= 0) {
	    a1[0] = 0;
	    for (int i = n-2; i >= 0; i--) {
	        if ((x1 & (1<<i)) != 0) a1[n-i-1] = 1;
	        else a1[n-i-1] = 0;
	    }
	} 
	else {
        a1[0] = 1;
        for (int i = n-2; i >= 0; i--) {
	        if ((abs(x1+1) & (1<<i)) != 0) a1[n-i-1] = 0;
	        else a1[n-i-1] = 1;
	    }
    }
	if (x2 >= 0) {
	    a2[0] = 0;
	    for (int i = n-2; i >= 0; i--) {
	        if ((x2 & (1<<i)) != 0) a2[n-i-1] = 1;
	        else a2[n-i-1] = 0;
	    }
	} 
	else {
        a2[0] = 1;
        for (int i = n-2; i >= 0; i--) {
	        if ((abs(x2+1) & (1<<i)) != 0) a2[n-i-1] = 0;
	        else a2[n-i-1] = 1;
	    }
    }
    for (int i = 0; i<n; i++) if (a1[i] != a2[i]) k++;

    printf("%d", k);
    return 0;
}
