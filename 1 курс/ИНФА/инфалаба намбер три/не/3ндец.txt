#include <stdio.h>
int main() {
	float x;
	int s = 0;
	scanf("%f", &x);
	int ax = fabs(x);
	float bx = fabs(x) - ax;
	while (trunc(ax) != 1) {
	    ax = ax >> 1;
	    s++;
	}
	int n = s + 15;
    if (x >= 0) {
        printf("0");
        for (int i=4; i>=0; i--) {
            if ((n & (1<<i)) != 0) printf("1");
	        else printf("0");
        }
    }
    else { 
        printf("1");
        for (int i=4; i>=0; i--) {
            if ((abs(n+1) & (1<<i)) != 0) printf("0");
	        else printf("1");
        }
    }
    ax = fabs(x);
    int d, k = 10;
    while (trunc(ax) != 1) {
        d = d & 1;
        printf("%d", d);
	    ax = ax >> 1;
	    k--;
    }
    int dx;
    while (bx != 1.0) {
        bx = bx * 2;
        dx = (int) bx;
        printf("%d", dx);
        k--;
    }
    while (k > 0) {
        k--;
        printf("0");
    }
    return 0;
    
}
