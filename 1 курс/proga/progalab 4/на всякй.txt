const int max_m = 7;

// определяем структуру для узла
typedef int t;
typedef struct node_tag {
    t value; // значение
    struct node_tag *next; // указатель на следующий узел
} node;

// вставка (push)
void push(node **head, t value) {
    node *tmp = malloc(sizeof(node));
    tmp->next = *head;
    tmp->value = value;
    *head = tmp;
}

// достать элемент сверху (pop)
t pop(node **head) {
    node *out;
    t value;
    out = *head;
    *head = (*head)->next;
    value = out->value;
    free(out);
    return value;
}

void clear(node **head) {
    node *out;
    t value;
    out = *head;
    *head = (*head)->next;
    value = out->value;
    free(out);
}

bool isEmpty(node **head) {
    if (getsize(head) == 0) return true;
    else return false;
}
// вывод стека (display)
void display(const node* head) {
    printf("Стек:\n");
    while (head) {
            printf("%d\n", head->value);
            head = head->next;
    }
}

// размер стека
t getsize(const node *head) {
    t size = 0;
    while (head) {
            size++;
            head = head->next;
    }
    return size;
}

void push_mid(node **head, t value) {
    t mid = (*head)->value / 2;
    t elem;
    node *tmp = NULL;
    for (int i = 0; i < mid; i++) {
        elem = pop(&head);
        push(&tmp, elem);
    }
    push(&head, value);
    for (int i = 0; i < mid; i++) {
        elem = pop(&tmp);
        push(&head, elem);
    }
}

// стрелочка
void arrow(int realpos, int arrowpos) { 
    if (realpos == arrowpos) printf(" >>> ");
    else printf("     ");
}
